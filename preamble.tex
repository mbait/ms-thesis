% G7-32 already loads: fontspec, polyglossia (Russian/English), amsmath,
% caption (GOST settings), geometry (GOST margins), titlesec, tikz,
% multirow, etoolbox. Do not reload them here.

% Monospace font (fontspec is loaded by G7-32)
\setmonofont{Nimbus Mono PS}

% Widen right margin: G7-32 default is 10 mm, GOST 7.0.11 requires 15 mm
\geometry{right=15mm}

% Bibliography
\usepackage[numbers,sort&compress]{natbib}

% Mathematics
\usepackage{amsfonts}
\usepackage{amssymb}

% Graphics
\usepackage{graphicx}
\usepackage{float}

% Tables
\usepackage{booktabs}
\usepackage{array}

% TikZ (not loaded by G7-32 class itself, only by preamble.inc which we don't use)
\usepackage{xcolor}
\usepackage{tikz}
\usetikzlibrary{arrows.meta, shapes.geometric, positioning, calc, fit, backgrounds}

% Listings (pseudocode only; no syntax highlighting)
\usepackage{listings}
\renewcommand\lstlistingname{Листинг}
\renewcommand\lstlistlistingname{Листинги}
\lstset{
  basicstyle=\small\ttfamily,
  breaklines=true,
  frame=single,
  numbers=none,
  showstringspaces=false,
}
\lstdefinestyle{pseudocode}{
  language={},
  basicstyle=\small\rmfamily,
  keywordstyle=\bfseries,
  frame=single,
  numbers=none,
  breaklines=false,      % lines are pre-wrapped by hand; auto-break miscomputes
  mathescape=true,       % width with a proportional font and breaks lines early
  columns=fullflexible,  % proportional font: let glyphs use natural widths
  keepspaces=true,       % do not collapse the spaces between words
}

% Prevent Latin technical terms from overflowing the line
\usepackage{microtype}

% Hyperlinks (load last)
\usepackage{hyperref}
\hypersetup{
  colorlinks=true,
  linkcolor=black,
  citecolor=black,
  urlcolor=blue,
}
