% --------------------------------------------------------
% Chapter and section headings (ГОСТ-style, no titlesec)
% --------------------------------------------------------
\makeatletter
% Numbered chapter: "1 НАЗВАНИЕ" — centred, bold, 16pt
\renewcommand{\@makechapterhead}[1]{%
  \vspace*{10\p@}%
  {\parindent \z@ \centering \normalfont
   \fontsize{16}{20}\bfseries
   \thechapter\enspace\MakeUppercase{#1}%
   \par\nobreak\vskip 20\p@}%
}
% Unnumbered chapter (ВВЕДЕНИЕ, ЗАКЛЮЧЕНИЕ, etc.)
\renewcommand{\@makeschapterhead}[1]{%
  \vspace*{10\p@}%
  {\parindent \z@ \centering \normalfont
   \fontsize{16}{20}\bfseries
   \MakeUppercase{#1}%
   \par\nobreak\vskip 20\p@}%
}
% Remove chapter prefix ("Chapter", "Глава", etc.)
\renewcommand{\chaptername}{}
\makeatother

% Section and subsection: bold, indent from paragraph margin
\makeatletter
\renewcommand{\section}{%
  \@startsection{section}{1}{\parindent}%
    {3.5ex \@plus 1ex \@minus .2ex}%
    {2.3ex \@plus .2ex}%
    {\normalfont\bfseries}}
\renewcommand{\subsection}{%
  \@startsection{subsection}{2}{\parindent}%
    {3.25ex \@plus 1ex \@minus .2ex}%
    {1.5ex \@plus .2ex}%
    {\normalfont\bfseries}}
\makeatother

% --------------------------------------------------------
% Table of contents — add dots for chapters
% --------------------------------------------------------
\makeatletter
\renewcommand{\l@chapter}[2]{%
  \addpenalty{-\@highpenalty}%
  \vskip 1.0em \@plus\p@
  \setlength\@tempdima{1.5em}%
  \begingroup
    \parindent \z@ \rightskip \@pnumwidth
    \parfillskip -\@pnumwidth
    \leavevmode \bfseries
    \advance\leftskip\@tempdima
    \hskip -\leftskip
    #1\nobreak%
    \leaders\hbox{$\m@th\mkern 4.5mu.\mkern 4.5mu$}\hfill%
    \nobreak\hb@xt@\@pnumwidth{\hss #2}\par
    \penalty\@highpenalty
  \endgroup}
\makeatother
